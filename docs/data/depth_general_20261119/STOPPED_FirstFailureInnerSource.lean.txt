import Recaman.SFinalWindowLength

namespace Recaman.FirstFailureInnerSource

open Recaman.SingleWindowEquality Recaman.SFinalWindowLength
open Recaman.CapacityEqualityInduction

/-- A short source at a first zero deficit, away from the initial boundary,
forces an earlier original actual A with no three-clock source at all. -/
theorem short_failure_has_inner_long_source (g : Nat → Bool) (u : Nat)
    (hu : 0<u) (hw : Win g u 3)
    (hpos : ∀ k, 0<k → k≤u+3 →
      cN (EndN g 0) 0 k+1≤cN (fun i => g i=false) 0 k)
    (hzero : cN (EndN g 0) 0 (u+4)=cN (fun i => g i=false) 0 (u+4)) :
    ∃ v, (v=u+1 ∨ v=u+2) ∧ EndN g 0 v ∧
      ∀ s k, s+k=v → Win g s k → 7≤k := by
  obtain ⟨hS,_,_⟩ := win3 g u hw
  have hinner : EndN g 0 (u+1) ∨ EndN g 0 (u+2) := by
    by_cases h1 : EndN g 0 (u+1)
    · exact Or.inl h1
    by_cases h2 : EndN g 0 (u+2)
    · exact Or.inr h2
    have hp := hpos u hu (by omega)
    have he0 := cN_succ_neg (EndN g 0) 0 u (by
      intro he
      have ha := he.1
      rw [Nat.zero_add,hS] at ha
      cases ha)
    have he1 := cN_succ_neg (EndN g 0) 0 (u+1) (by simpa only [Nat.zero_add] using h1)
    have he2 := cN_succ_neg (EndN g 0) 0 (u+2) (by simpa only [Nat.zero_add] using h2)
    have he3 := cN_succ_le (EndN g 0) 0 (u+3)
    have hs0 := cN_succ_pos (fun i => g i=false) 0 u (by simpa only [Nat.zero_add] using hS)
    have hs := count_prefix_le (fun i => g i=false) 0 (u+1) (u+4) (by omega)
    omega
  rcases hinner with hv | hv
  · refine ⟨u+1,Or.inl rfl,hv,?_⟩
    intro s k hsk hwin
    rcases window_three_or_at_least_seven g s k hwin with h3 | h7
    · subst k
      have ha := (win3 g s hwin).2.2
      rw [show s+2=u by omega,hS] at ha
      cases ha
    · exact h7
  · refine ⟨u+2,Or.inr rfl,hv,?_⟩
    intro s k hsk hwin
    rcases window_three_or_at_least_seven g s k hwin with h3 | h7
    · subst k
      have ha := (win3 g s hwin).2.1
      rw [show s+1=u by omega,hS] at ha
      cases ha
    · exact h7

end Recaman.FirstFailureInnerSource
