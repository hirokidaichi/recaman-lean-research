import Recaman.SingleWindowEquality

namespace Recaman.GeometricKernelCertificate

/-- The finite coefficient of one lower addition after the weighted
height sum, as a recurrence in its distance from the top level. -/
def kernel : Nat → Int
  | 0 => 0
  | 1 => 0
  | n+2 => 3*kernel (n+1)+3*(2*((n+1 : Nat) : Int)-1)

theorem kernel_exact (N : Nat) :
    kernel (N+1)=(3 : Int)^(N+1)-3*((N+1 : Nat) : Int) := by
  induction N with
  | zero => decide
  | succ n ih =>
    rw [show n+1+1=n+2 by omega,kernel,ih,
      show n+2=(n+1)+1 by omega,Int.pow_succ,Int.natCast_add]
    have hone : ((1 : Nat) : Int)=1 := rfl
    rw [hone]
    grind

theorem kernel_bound (N : Nat) : 0≤kernel N ∧ kernel N≤(3 : Int)^N := by
  induction N with
  | zero => simp [kernel]
  | succ n ih =>
    constructor
    · cases n with
      | zero => decide
      | succ n =>
        rw [kernel]
        have := ih.1
        have hcast : (0 : Int)≤(n : Int) := by omega
        omega
    · rw [kernel_exact]
      omega

def coefficient (N : Nat) : Nat → Int
  | 0 => 0
  | d+1 => coefficient N d+
      (if 2≤d+1 then (2*((d+1 : Nat) : Int)-3)*(3 : Int)^(N-(d+1)+1) else 0)

/-- The explicit finite coefficient has the same recurrence at its full
distance. This certificate concerns only the arithmetic kernel. -/
theorem partial_coefficient (N d : Nat) (hd : d≤N) :
    coefficient N d=(3 : Int)^(N-d)*kernel d := by
  induction d with
  | zero => simp [coefficient,kernel]
  | succ d ih =>
    cases d with
    | zero => simp [coefficient,kernel]
    | succ d =>
      have hi := ih (by omega)
      rw [coefficient,if_pos (by omega),hi,kernel,
        show N-(d+1)=N-(d+1+1)+1 by omega,Int.pow_succ]
      have hlen : (((d+1+1 : Nat) : Int))=(((d+1 : Nat) : Int))+1 := by omega
      have he : 2*((d+1+1 : Nat) : Int)-3=2*((d+1 : Nat) : Int)-1 := by omega
      rw [he]
      grind

theorem full_coefficient_bound (N : Nat) :
    0≤coefficient N N ∧ coefficient N N≤(3 : Int)^N := by
  rw [partial_coefficient N N (Nat.le_refl N)]
  simp only [Nat.sub_self,Int.pow_zero,Int.one_mul]
  exact kernel_bound N

#print axioms full_coefficient_bound

end Recaman.GeometricKernelCertificate
