private theorem split_at_addition (f : Nat → Bool) (n K t : Nat)
    (hnt : n≤t) (ht : t<n+K) (ha : f t=true) :
    seg f n K=seg f n (t-n)++[true]++seg f (t+1) (n+K-(t+1)) := by
  have h:=seg_add f n (t-n+1) (n+K-(t+1))
  rw [show t-n+1+(n+K-(t+1))=K by omega,seg_succ,
    show n+(t-n)=t by omega,ha,show n+(t-n+1)=t+1 by omega] at h
  exact h

private theorem SAA_ending (f : Nat → Bool) (n t : Nat) (hnt : n≤t) (v : List Bool)
    (hv : seg f n (t-n)=v++[false,true,true]) :
    n+3≤t ∧ f (t-3)=false ∧ f (t-2)=true ∧ f (t-1)=true := by
  let x:=seg f n (t-n)
  change x=v++[false,true,true] at hv
  have hlen : v.length+3=t-n := by
    have h:=congrArg List.length hv
    rw [show x.length=t-n from seg_length f n (t-n),List.length_append] at h
    simpa using h.symm
  have hcut:=seg_drop f n (t-n) v.length (by omega)
  change x.drop v.length=seg f (n+v.length) (t-n-v.length) at hcut
  rw [hv,List.drop_left,show n+v.length=t-3 by omega,show t-n-v.length=3 by omega] at hcut
  have h0:=seg_get f (t-3) 3 0 (by decide)
  have h1:=seg_get f (t-3) 3 1 (by decide)
  have h2:=seg_get f (t-3) 3 2 (by decide)
  rw [← hcut,show t-3+0=t-3 by omega] at h0
  rw [← hcut,show t-3+1=t-2 by omega] at h1
  exact ⟨by omega,(Option.some.inj h0).symm,(Option.some.inj h1).symm,(Option.some.inj h2).symm⟩
  exact ⟨by omega,by omega,(Option.some.inj h0).symm,(Option.some.inj h1).symm,(Option.some.inj h2).symm⟩

/-- The local clause retains the exact three signs immediately preceding
its endpoint in the original segment. -/
theorem low_cross_local_cut (f : Nat → Bool) (n K t : Nat)
    (hnt : n≤t) (ht : t<n+K) (ha : f t=true)
    (hQ : cross (seg f n K)≤20) (hmod : cross (seg f n K)%8=4)
    (hS : subtractions f n K=6)
    (hl : Recaman.ThreeNonsuppliedData.localSupply (seg f n (t-n))=true) :
    (subtractions f n (t-n)=5 ∨ subtractions f n (t-n)=6) ∧
    n+3≤t ∧ f (t-3)=false ∧ f (t-2)=true ∧ f (t-1)=true := by
  let x:=seg f n (t-n)
  let z:=seg f (t+1) (n+K-(t+1))
  have hs : seg f n K=x++[true]++z := split_at_addition f n K t hnt ht ha
  have hc : cntF (seg f n K)=6 := by rw [cntF_seg]; exact hS
  rw [hs,cntF_append,cntF_append,show cntF [true]=0 by decide,Nat.add_zero] at hc
  rw [hs] at hQ hmod
  obtain ⟨⟨v,hv⟩,hz⟩:=local_source_at_last_two_S x z hQ hmod hl
  have hx : cntF x=subtractions f n (t-n) := cntF_seg f n (t-n)
  exact ⟨by omega,SAA_ending f n t hnt v hv⟩

/-- The old clause at small cross has all six subtractions before it. -/
theorem low_cross_old_cut (f : Nat → Bool) (n K t : Nat) (w : List Bool)
    (hnt : n≤t) (ht : t<n+K) (ha : f t=true) (hw : BlockWord w)
    (hQ : cross (seg f n K)≤20) (hS : subtractions f n K=6)
    (ho : tailOldRay w (seg f n (t-n)) 0) : subtractions f n (t-n)=6 := by
  let x:=seg f n (t-n)
  let z:=seg f (t+1) (n+K-(t+1))
  have hs : seg f n K=x++[true]++z := split_at_addition f n K t hnt ht ha
  have hc : cntF (seg f n K)=6 := by rw [cntF_seg]; exact hS
  rw [hs,cntF_append,cntF_append,show cntF [true]=0 by decide,Nat.add_zero] at hc
  rw [hs] at hQ
  have hz:=Recaman.LowCrossOldClocks.low_cross_old_after_last_S w x z hw hc hQ ho
  have hx : cntF x=subtractions f n (t-n) := cntF_seg f n (t-n)
  omega

/-- Two different SAA endpoints cannot have the same subtraction rank. -/
theorem SAA_endpoints_rank_strict (f : Nat → Bool) (n a b : Nat) (hab : a<b)
    (ha : n+3≤a) (hb : n+3≤b)
    (ha2 : f (a-2)=true) (ha1 : f (a-1)=true) (hb3 : f (b-3)=false) :
    subtractions f n (a-n)<subtractions f n (b-n) := by
  have hba : a≤b-3 := by
    by_cases h : a≤b-3
    · exact h
    · have he : b-3=a-2 ∨ b-3=a-1 := by omega
      rcases he with he|he
      · rw [he,ha2] at hb3; cases hb3
      · rw [he,ha1] at hb3; cases hb3
  have h1 : subtractions f n (a-n)≤subtractions f n (b-3-n) :=
    count_prefix_le (fun x=>f x=false) n (a-n) (b-3-n) (by omega)
  have h2 : subtractions f n (b-3-n+1)=subtractions f n (b-3-n)+1 :=
    cN_succ_pos _ n (b-3-n) (by rw [show n+(b-3-n)=b-3 by omega]; exact hb3)
  have h3 : subtractions f n (b-3-n+1)≤subtractions f n (b-n) :=
    count_prefix_le (fun x=>f x=false) n (b-3-n+1) (b-n) (by omega)
  omega

/-- The seven-pair local source cut records the original indices.
The exceptional source ends at the single position n+11. -/
theorem cross28_local_segment_cut (f : Nat → Bool) (n K t : Nat)
    (hnt : n≤t) (ht : t<n+K) (ha : f t=true)
    (hQ : cross (seg f n K)=28) (hS : subtractions f n K=6)
    (hl : Recaman.ThreeNonsuppliedData.localSupply (seg f n (t-n))=true) :
    t=n+11 ∨
      (4≤subtractions f n (t-n) ∧ subtractions f n (t-n)≤6 ∧
      (subtractions f n (t-n)=4 → additions f n (t-n)=2) ∧
      n+3≤t ∧ f (t-3)=false ∧ f (t-2)=true ∧ f (t-1)=true) := by
  let x:=seg f n (t-n)
  let z:=seg f (t+1) (n+K-(t+1))
  have hs : seg f n K=x++[true]++z := split_at_addition f n K t hnt ht ha
  have hc : cntF (seg f n K)=6 := by rw [cntF_seg]; exact hS
  rw [hs,cntF_append,cntF_append,show cntF [true]=0 by decide,Nat.add_zero] at hc
  rw [hs] at hQ
  rcases cross28_local_source x z hQ hc hl with h|h
  · obtain ⟨⟨v,hv⟩,hz,ha2⟩:=h
    have hx : cntF x=subtractions f n (t-n) := cntF_seg f n (t-n)
    refine Or.inr ⟨by omega,by omega,?_,SAA_ending f n t hnt v hv⟩
    intro h4
    have hA:=ha2 (by omega)
    rw [show x=seg f n (t-n) by rfl,ones_seg_eq_count] at hA
    exact_mod_cast hA
  · have hlen : x.length=t-n := seg_length f n (t-n)
    exact Or.inl (by omega)

theorem cross28_old_segment_cut (f : Nat → Bool) (n K t : Nat) (w : List Bool)
    (hnt : n≤t) (ht : t<n+K) (ha : f t=true) (hw : BlockWord w)
    (hQ : cross (seg f n K)=28) (hS : subtractions f n K=6)
    (ho : tailOldRay w (seg f n (t-n)) 0) :
    t=n+9 ∧ subtractions f n (t-n)=5 ∧ additions f n (t-n)=4 := by
  let x:=seg f n (t-n)
  let z:=seg f (t+1) (n+K-(t+1))
  have hs : seg f n K=x++[true]++z := split_at_addition f n K t hnt ht ha
  have hc : cntF (seg f n K)=6 := by rw [cntF_seg]; exact hS
  rw [hs,cntF_append,cntF_append,show cntF [true]=0 by decide,Nat.add_zero] at hc
  rw [hs] at hQ
  obtain ⟨hs5,ha4⟩:=Recaman.LowCrossOldClocks.cross28_old_cut w x z hw hc hQ ho
  change cntF (seg f n (t-n))=5 at hs5
  change ones (seg f n (t-n))=4 at ha4
  rw [cntF_seg] at hs5
  rw [ones_seg_eq_count] at ha4
  have hc' := (cN_walk f n (t-n)).2
  exact ⟨by omega,hs5,by omega⟩

theorem cross_two_addition_cuts (f : Nat → Bool) (n K a b : Nat)
    (hna : n≤a) (hab : a<b) (hb : b<n+K) (haa : f a=true) (hab' : f b=true) :
    4*((additions f n (a-n) : Int)+1)*
      ((subtractions f n (b-n) : Int)-subtractions f n (a-n)) +
    4*((additions f n (b-n) : Int)+1)*
      ((subtractions f n K : Int)-subtractions f n (b-n)) ≤ cross (seg f n K) := by
  let x:=seg f n (a-n)
  let y:=seg f (a+1) (b-(a+1))
  let z:=seg f (b+1) (n+K-(b+1))
  have hsplit : seg f n K=seg f n (b-n)++[true]++z :=
    split_at_addition f n K b (by omega) hb hab'
  have hpre : seg f n (b-n)=x++[true]++y := by
    have h:=split_at_addition f n (b-n) a hna (by omega) haa
    rw [show n+(b-n)=b by omega] at h
    exact h
  have hy : (cntF y : Int)=(subtractions f n (b-n) : Int)-subtractions f n (a-n) := by
    have h:=congrArg cntF hpre
    rw [cntF_append,cntF_append,show cntF [true]=0 by decide,Nat.add_zero,cntF_seg] at h
    have hx : cntF x=subtractions f n (a-n) := cntF_seg f n (a-n)
    omega
  have hz : (cntF z : Int)=(subtractions f n K : Int)-subtractions f n (b-n) := by
    have h:=congrArg cntF hsplit
    rw [cntF_append,cntF_append,show cntF [true]=0 by decide,Nat.add_zero,cntF_seg,cntF_seg] at h
    omega
  have hx : ones x=(additions f n (a-n) : Int) := ones_seg_eq_count f n (a-n)
  rw [hsplit,cross_at_addition,ones_seg_eq_count,← hz,hpre,cross_at_addition,hx,← hy]
  have h0:=cross_nonnegative x
  have h1:=cross_nonnegative y
  have h2:=cross_nonnegative z
  omega

end Recaman.LowCrossSegmentCuts
