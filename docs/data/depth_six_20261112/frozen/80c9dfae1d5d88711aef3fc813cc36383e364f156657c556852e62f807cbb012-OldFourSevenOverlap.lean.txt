namespace Recaman.OldFourSevenOverlap

open Recaman.OldFourHighWindowSpectrum Recaman.SevenWindowPrefixFive
open Recaman.DepthSixGapResidual Recaman.BlockSuffixPositivity
open Recaman.SingleWindowEquality Recaman.CapacityEqualityInduction

/-- With no high free addition, an original old-source gap-four counter
internally forces the overlapping word SASAASAA and two earlier actual
suppliers at the same target height. -/
theorem old_four_zero_high_seven_overlap (f : Nat → Bool) (n a b t : Nat)
    {w : List Bool} (hw : BlockWord w) (hpre : seg f 0 n=w++List.replicate a true)
    (hnb : n<b) (hfloor : ∀ i, n≤i → i≤b → Fw f n-6≤Fw f i)
    (hpos : ∀ k, 0<k → k≤b-n → cN (EndN f 0) n k+1≤cN (fun i => f i=false) n k)
    (hS : ∀ i, b≤i → i<b+4 → f i=false) (he : EndN f 0 (b+4))
    (hcounter : cN (fun i => f i=true ∧ ¬EndN f 0 i) n (b-n)≤4)
    (htn : t<n) (hwin : Win f t (b+4-t))
    (hZero : cN (fun i => Fw f n-4≤Fw f i ∧ f i=true ∧ ¬EndN f 0 i) n (b-n)=0) :
    ∃ v, n≤v ∧ v+8<b ∧ Fw f v=Fw f n ∧ Win f v 7 ∧
      seg f v 8=[false,true,false,true,true,false,true,true] ∧
      Fw f (v+1)=Fw f n-1 ∧ Fw f (v+3)=Fw f n-1 ∧
      EndN f 0 (v+1) ∧ EndN f 0 (v+3) := by
  have hActual (x : Nat) (hnx : n≤x) (hxb : x<b) (hxF : Fw f n-4≤Fw f x)
      (hxA : f x=true) : EndN f 0 x := by
    by_cases h : EndN f 0 x
    · exact h
    · exact (not_of_cN_zero
        (fun i => Fw f n-4≤Fw f i ∧ f i=true ∧ ¬EndN f 0 i) n (b-n)
        hZero x hnx (by omega) ⟨hxF,hxA,h⟩).elim
  obtain ⟨_,_,_,hTarget,_⟩ := positive_depth_six_counter_profile f n a b 4 hw hpre
    hnb hfloor hpos hS he hcounter
  have hDown := fw_false_run f b 4 hS
  obtain ⟨q,hnq,hqb,hq,hqA⟩ := ivt f b (Fw f n+3) (by omega) (b-n) n (by omega) (by omega)
  have hqe := hActual q hnq hqb (by omega) hqA
  obtain ⟨u,hnu,huq,huH,hWin,hCases⟩ := old_four_high_source_spectrum f n a b t q 3 hw hpre
    hnb hfloor hpos hS he hcounter htn hwin hnq hqb (by omega) (by omega) hqe
  have h3 : q-u=3 := by
    rcases hCases with h | h
    · exact h.1
    · have := h.2.1
      omega
  have hWin3 : Win f u 3 := by rwa [h3] at hWin
  obtain ⟨hSu,hA1,hA2⟩ := win3 f u hWin3
  have hPrevA : f (q-1)=true := by rwa [show q-1=u+2 by omega]
  have hPrevF : Fw f (q-1)=Fw f n+1 := by
    rcases fw_cases f (q-1) with ⟨h,_⟩ | ⟨_,h⟩
    · rw [show q-1+1=q by omega] at h
      omega
    · simp [hPrevA] at h
  have hPrevE := hActual (q-1) (by omega) (by omega) (by omega) hPrevA
  obtain ⟨v,hnv,hvq,hvH,hInner,hCasesV⟩ := old_four_high_source_spectrum f n a b t (q-1) 2 hw hpre
    hnb hfloor hpos hS he hcounter htn hwin (by omega) (by omega) (by omega) (by omega) hPrevE
  have h7 : q-1-v=7 := by
    rcases hCasesV with h | h
    · have hV3 := h.1
      have hVWin : Win f v 3 := by rwa [hV3] at hInner
      have hVA := (win3 f v hVWin).2.1
      rw [show v+1=u by omega] at hVA
      simp [hSu] at hVA
    · exact h.1
  have hWin7 : Win f v 7 := by rwa [h7] at hInner
  have hWord7 := actual_seven_ending_SA f v hWin7
    (by rwa [show v+5=u by omega]) (by rwa [show v+6=u+1 by omega])
  have hWord : seg f v 8=[false,true,false,true,true,false,true,true] := by
    rw [show 8=7+1 by rfl,seg_succ,hWord7,show v+7=q-1 by omega,hPrevA]
    rfl
  have hV0 := seg_get f v 8 0 (by decide)
  have hV1 := seg_get f v 8 1 (by decide)
  have hV2 := seg_get f v 8 2 (by decide)
  have hV3 := seg_get f v 8 3 (by decide)
  rw [hWord] at hV0 hV1 hV2 hV3
  have hs0 : f v=false := by exact (Option.some.inj hV0).symm
  have ha1 : f (v+1)=true := by exact (Option.some.inj hV1).symm
  have hs2 : f (v+2)=false := by exact (Option.some.inj hV2).symm
  have ha3 : f (v+3)=true := by exact (Option.some.inj hV3).symm
  have hF1 : Fw f (v+1)=Fw f n-1 := by
    rcases fw_cases f v with ⟨_,h⟩ | ⟨h,_⟩
    · simp [hs0] at h
    · omega
  have hF2 : Fw f (v+2)=Fw f n := by
    rcases fw_cases f (v+1) with ⟨h,_⟩ | ⟨_,h⟩
    · rw [show v+1+1=v+2 by omega] at h
      omega
    · simp [ha1] at h
  have hF3 : Fw f (v+3)=Fw f n-1 := by
    rcases fw_cases f (v+2) with ⟨_,h⟩ | ⟨h,_⟩
    · simp [hs2] at h
    · rw [show v+2+1=v+3 by omega] at h
      omega
  exact ⟨v,hnv,by omega,by omega,hWin7,hWord,hF1,hF3,
    hActual (v+1) (by omega) (by omega) (by omega) ha1,
    hActual (v+3) (by omega) (by omega) (by omega) ha3⟩

end Recaman.OldFourSevenOverlap
