
def endpointWords : List (List Bool) :=
  Recaman.ThreeNonsuppliedData.endpointWords ++
  [[false,true,true,false,true,true,true],
   [false,true,false,true,true,true,true],
   [false,true,false,true,true,false,true,true,true],
   [false,false,true,true,true,true,true],
   [false,false,true,true,true,true,false,true,true],
   [false,false,true,true,true,false,true,true,true],
   [false,false,false,true,true,true,true,false,true,true,true],
   [false,false,false,false,true,true,true,true,false,true,true,true,true]]

def endpointGood (s : State) : Bool :=
  decide (((s.subtractions : Int)-(s.additions : Int)+(s.nonsupplied : Int)=1) → s.word ∈ endpointWords)

def goodState (s : State) : Bool :=
  decide (s.additions≤10 ∧ s.subtractions≤19 ∧ s.nonsupplied≤4 ∧ gap s 0=s) &&
    (List.range 20).all (fun j => endpointGood (gap s j))

abbrev Row := Nat × Nat

structure Table where
  size : Nat
  stateCode : Nat → Nat
  edgeCode : Nat → Nat

def unpack (code : Nat) : State :=
  ⟨Recaman.ThreeNonsuppliedData.decodeWord 29 (code/4096),code%16,(code/16)%32,(code/512)%8⟩

def stateAt (rs : Table) (i : Nat) : State := unpack (rs.stateCode i)

def edgeAt (payload : Nat) (edge : Nat) : Nat := (payload >>> (14*edge)) % 16384

def checkOption (rs : Table) (s : Option State) (j : Nat) : Bool :=
  match s with
  | none => decide (j=0)
  | some t => decide (0<j ∧ j-1<rs.size ∧ stateAt rs (j-1)=t)

def checkOne (rs : Table) (i : Nat) : Bool :=
  let payload := rs.edgeCode i
  let s := stateAt rs i
  [goodState s,(List.range 20).all (fun j =>
    checkOption rs (macroA s j false) (edgeAt payload (2*j)) &&
    checkOption rs (macroA s j true) (edgeAt payload (2*j+1)))].all id

def verified (rs : Table) : Bool := (List.range rs.size).all (checkOne rs)

def verifiedRange (rs : Table) (lo count : Nat) : Bool :=
  (List.range count).all fun j => checkOne rs (lo+j)

def Member (rs : Table) (s : State) : Prop := ∃ i, i<rs.size ∧ stateAt rs i=s

def Covered (rs : Table) (s : State) : Prop := ∃ t j, Member rs t ∧ gap t j=s

theorem verifiedRange_at (rs : Table) (lo count : Nat) (h : verifiedRange rs lo count=true)
    (i : Nat) (hlo : lo≤i) (hhi : i<lo+count) : checkOne rs i=true := by
  have ht := List.all_eq_true.mp h (i-lo) (List.mem_range.mpr (by omega))
  rwa [Nat.add_sub_cancel' hlo] at ht

theorem row_checked (rs : Table) (h : verified rs=true) (i : Nat) (hi : i<rs.size) :
    checkOne rs i=true := List.all_eq_true.mp h i (List.mem_range.mpr hi)

theorem member_good (rs : Table) (h : verified rs=true) (s : State) (hs : Member rs s) :
    goodState s=true := by
  obtain ⟨i,hi,rfl⟩ := hs
  have hr := row_checked rs h i hi
  simp only [checkOne,List.all_cons,List.all_nil,id_eq,Bool.and_true,Bool.and_eq_true] at hr
  exact hr.1

theorem member_additions_le_ten (rs : Table) (h : verified rs=true) (s : State) (hs : Member rs s) :
    s.additions≤10 := by
  have hg := member_good rs h s hs
  simp only [goodState,Bool.and_eq_true,decide_eq_true_eq] at hg
  exact hg.1.1

theorem member_covered (rs : Table) (h : verified rs=true) (s : State) (hs : Member rs s) :
    Covered rs s := by
  have hg := member_good rs h s hs
  simp only [goodState,Bool.and_eq_true,decide_eq_true_eq] at hg
  exact ⟨s,0,hs,hg.1.2.2.2⟩

theorem member_macroA (rs : Table) (h : verified rs=true) (s t : State)
    (hs : Member rs s) (j : Nat) (hj : j<20) (e : Bool) (ht : macroA s j e=some t) : Member rs t := by
  obtain ⟨i,hi,rfl⟩ := hs
  have hr := row_checked rs h i hi
  simp only [checkOne,List.all_cons,List.all_nil,id_eq,Bool.and_true,Bool.and_eq_true] at hr
  have hc := List.all_eq_true.mp hr.2 j (List.mem_range.mpr hj)
  simp only [Bool.and_eq_true] at hc
  have he : checkOption rs (macroA (stateAt rs i) j e)
      (edgeAt (rs.edgeCode i) (2*j+(if e then 1 else 0)))=true := by
    cases e
    · simpa using hc.1
    · simpa using hc.2
  rw [ht] at he
  have hh := of_decide_eq_true he
  exact ⟨_,hh.2.1,hh.2.2⟩

theorem covered_additions_le_ten (rs : Table) (h : verified rs=true) (s : State) (hs : Covered rs s) :
    s.additions≤10 := by
  obtain ⟨t,j,hm,rfl⟩ := hs
  exact member_additions_le_ten rs h t hm

theorem covered_stepS (rs : Table) (s : State) (hs : Covered rs s) : Covered rs (stepS s) := by
  obtain ⟨t,j,hm,rfl⟩ := hs
  exact ⟨t,j+1,hm,(gap_add t j 1).symm⟩

theorem covered_stepA (rs : Table) (h : verified rs=true) (s t : State)
    (hs : Covered rs s) (e : Bool) (ht : stepA s e=some t) : Covered rs t := by
  obtain ⟨u,j,hm,hgap⟩ := hs
  have hmstep : macroA u (min j 19) e=some t := by
    unfold macroA
    rw [←gap_clamp,hgap]
    exact ht
  exact member_covered rs h t (member_macroA rs h u t hm (min j 19) (by omega) e hmstep)

theorem covered_endpoint_word (rs : Table) (h : verified rs=true) (s : State)
    (hs : Covered rs s)
    (hd : (s.subtractions : Int)-(s.additions : Int)+(s.nonsupplied : Int)=1) : s.word ∈ endpointWords := by
  obtain ⟨t,j,hm,rfl⟩ := hs
  have hg := member_good rs h t hm
  simp only [goodState,Bool.and_eq_true,decide_eq_true_eq] at hg
  have hj : min j 19<20 := by omega
  have hc := List.all_eq_true.mp hg.2 (min j 19) (List.mem_range.mpr hj)
  have hp := of_decide_eq_true hc
  rw [←gap_clamp] at hp
  exact hp hd
