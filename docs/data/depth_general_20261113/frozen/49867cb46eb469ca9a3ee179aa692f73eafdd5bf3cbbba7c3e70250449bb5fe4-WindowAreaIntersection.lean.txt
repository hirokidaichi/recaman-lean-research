import Recaman.TwoWindowAreaOverlap

namespace Recaman.WindowAreaIntersection

open Recaman.TwoWindowAreaOverlap Recaman.HighWindowSquareBudget Recaman.LevelTriangleArea
open Recaman.SingleWindowEquality

/-- This interval resource is nonnegative without any depth or floor bound. -/
theorem surplus_nonnegative (f : Nat → Bool) (s : Nat) (r : Int) (K : Nat) :
    0≤areaSurplus f s r K := by
  have h := (signed_area_with_low_weights f s r K).2
  change _≤_ at h
  unfold areaSurplus
  omega

theorem surplus_split (f : Nat → Bool) (s t b : Nat) (r : Int)
    (hst : s≤t) (htb : t≤b) :
    areaSurplus f s r (b-s)=areaSurplus f s r (t-s)+areaSurplus f t r (b-t) := by
  have hw := low_weights_append f s r (t-s) (b-t)
  rw [show t-s+(b-t)=b-s by omega,show s+(t-s)=t by omega] at hw
  unfold areaSurplus
  rw [show s+(b-s)=b by omega,show s+(t-s)=t by omega,show t+(b-t)=b by omega,
    area_from_level_split f s t b r hst htb,hw]
  omega

theorem surplus_subinterval (f : Nat → Bool) (s t q b : Nat) (r : Int)
    (hst : s≤t) (htq : t≤q) (hqb : q≤b) :
    areaSurplus f t r (q-t)≤areaSurplus f s r (b-s) := by
  have h1 := surplus_split f s t q r hst htq
  have h2 := surplus_split f s q b r (by omega) hqb
  have h3 := surplus_nonnegative f s r (t-s)
  have h4 := surplus_nonnegative f q r (b-q)
  omega

/-- An original Win pays its charge from its own interval, retaining
all low additions and the exact source height. -/
theorem actual_window_charge (f : Nat → Bool) (t q : Nat) (r : Int)
    (htq : t<q) (hr : r≤Fw f t) (hw : Win f t (q-t)) :
    windowCharge f r t q≤areaSurplus f t r (q-t) := by
  have hq : Fw f q=Fw f t+1 := by
    have h := hw.2.1
    rwa [show t+(q-t)=q by omega] at h
  have hbelow (x : Int) (hx : r≤x) : belowTriangle r x=0 := by
    unfold belowTriangle
    rw [Int.max_eq_left (show r-x≤0 by omega)]
    omega
  have htri : aboveTriangle r (Fw f t+1)-aboveTriangle r (Fw f t)=2*(Fw f t-r+1) := by
    unfold aboveTriangle
    rw [Int.max_eq_right (show 0≤Fw f t+1-r by omega),Int.max_eq_right (by omega : 0≤Fw f t-r)]
    ring
  have hnonneg := low_weights_nonnegative f t r (q-t)
  have hpoly : 2*(Fw f t-r)*(((q-t : Nat) : Int)-1)-2=
      2*((q-t : Nat) : Int)*(Fw f t-r)-2*(Fw f t-r+1) := by ring
  unfold areaSurplus areaFromLevel windowCharge
  rw [hw.2.2,show t+(q-t)=q by omega,hbelow _ hr,hbelow _ (by omega),hq,hpoly]
  omega

/-- For two interlaced original windows, their common interval is the
only repeated area resource. No bound on depth, run count or length is
imposed, and no replacement of low additions by free count is made. -/
theorem two_window_intersection_budget (f : Nat → Bool) (s t1 t2 q1 q2 b : Nat) (r : Int)
    (hs : s≤t1) (h12 : t1≤t2) (h2q : t2≤q1) (hqq : q1≤q2) (hqb : q2≤b)
    (h1 : t1<q1) (h2 : t2<q2) (hr1 : r≤Fw f t1) (hr2 : r≤Fw f t2)
    (hw1 : Win f t1 (q1-t1)) (hw2 : Win f t2 (q2-t2)) :
    windowCharge f r t1 q1+windowCharge f r t2 q2≤
      areaSurplus f s r (b-s)+areaSurplus f t2 r (q1-t2) := by
  have hc1 := actual_window_charge f t1 q1 r h1 hr1 hw1
  have hc2 := actual_window_charge f t2 q2 r h2 hr2 hw2
  have hleft := surplus_split f t1 t2 q1 r h12 h2q
  have hright := surplus_split f t1 t2 q2 r h12 (by omega)
  have hwhole := surplus_subinterval f s t1 q2 b r hs (by omega) hqb
  omega

end Recaman.WindowAreaIntersection
