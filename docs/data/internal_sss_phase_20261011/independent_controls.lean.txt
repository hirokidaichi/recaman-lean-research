-- Chronological period ASASASSSAAA.
def donor (x : Int) : Bool :=
  decide (x % 11 = 0 ∨ x % 11 = 2 ∨ x % 11 = 4 ∨ 8 ≤ x % 11)

theorem donor_periodic : ∀ x : Int, donor (x + 11) = donor x := by
  intro x
  simp [donor]

theorem donor_facts :
    past donor 11 11 = [true,true,true,false,false,false,true,false,true,false,true] ∧
    P2 (past donor 11 11) ∧ ssCount (past donor 11 11) = 2 ∧
    oldestOffset (past donor 11 11) = 10 ∧ donor 1 = false ∧
    (11 : Int) - 11 ≤ 1 ∧ (1 : Int) < 5 ∧ (5 : Int) + 2 < 11 ∧
    donor 5 = false ∧ donor 6 = false ∧ donor 7 = false ∧
    donor 10 = true ∧ P2 (past donor 10 3) ∧
    ssCount (past donor 10 3) = 0 ∧ endpointPhase 11 10 3 = 7 := by decide

theorem donor_minimal :
    ∀ j : Nat, 0 < j → j < 11 → ¬ P2 ((past donor 11 11).take j) := by
  intro j hj hj11
  have h : ∀ j : Fin 11, 0 < j.val → ¬ P2 ((past donor 11 11).take j.val) := by decide
  exact h ⟨j, hj11⟩ hj

-- Full unbounded statement obtained by applying the reviewed theorem.
theorem donor_result :
    phase 11 (11 - oldestOffset (past donor 11 11)) ≠ phase 11 5 ∧
      ∀ (t : Int) (l : Nat), donor t = true → P2 (past donor t l) →
        ssCount (past donor t l) ≤ 1 →
        endpointPhase 11 t l ≠ phase 11 (11 - oldestOffset (past donor 11 11)) ∧
        endpointPhase 11 t l ≠ phase 11 5 := by
  exact Recaman.SS2InternalSSS.donor_internal_sss_two_excluded_phases
    donor 11 (by decide) donor_periodic 11 11
    (by decide) (by decide) donor_minimal 5
    (by decide) (by decide) (by decide) (by decide) (by decide)

example : phase 11 1 ≠ phase 11 5 ∧
    endpointPhase 11 10 3 ≠ phase 11 1 ∧
    endpointPhase 11 10 3 ≠ phase 11 5 := by
  constructor
  · exact donor_result.1
  · exact donor_result.2 10 3 (by decide) (by decide) (by decide)

-- A wrapping, negative-mass separation instance.
def wrap (x : Int) : Bool := decide (x % 6 = 1 ∨ x % 6 = 5)

example : ∀ x : Int, wrap (x + 6) = wrap x := by
  intro x
  simp [wrap]

example : ssCount (past wrap 7 7) = 2 ∧ mass (past wrap 0 6) = -2 ∧
    wrap 0 = false ∧ wrap 2 = false ∧ wrap 3 = false ∧ wrap 4 = false ∧
    phase 6 0 ≠ phase 6 2 := by decide

def allS (_ : Int) : Bool := false

example : ∀ x : Int, allS (x + 1) = allS x := by
  intro x
  rfl

-- Removing the lower window bound alone.
example : ssCount (past allS 4 3) = 2 ∧
    ¬ ((4 : Int) - 3 ≤ 0) ∧ (0 : Int) < 1 ∧ (1 : Int) + 2 < 4 ∧
    allS 0 = false ∧ allS 1 = false ∧ allS 2 = false ∧ allS 3 = false ∧
    phase 1 0 = phase 1 1 := by decide

-- Removing strict s<q alone.
example : ssCount (past allS 3 3) = 2 ∧
    (3 : Int) - 3 ≤ 0 ∧ ¬ ((0 : Int) < 0) ∧ (0 : Int) + 2 < 3 ∧
    allS 0 = false ∧ allS 1 = false ∧ allS 2 = false ∧
    phase 1 0 = phase 1 0 := by decide

-- Removing triple containment alone.
example : ssCount (past allS 3 3) = 2 ∧
    (3 : Int) - 3 ≤ 0 ∧ (0 : Int) < 1 ∧ ¬ ((1 : Int) + 2 < 3) ∧
    allS 0 = false ∧ allS 1 = false ∧ allS 2 = false ∧ allS 3 = false ∧
    phase 1 0 = phase 1 1 := by decide

-- Removing the SS budget alone.
example : ssCount (past allS 4 4) = 3 ∧
    (4 : Int) - 4 ≤ 0 ∧ (0 : Int) < 1 ∧ (1 : Int) + 2 < 4 ∧
    allS 0 = false ∧ allS 1 = false ∧ allS 2 = false ∧ allS 3 = false ∧
    phase 1 0 = phase 1 1 := by decide

-- Chronological SAASSS has period six but fails the designated period three.
def sixSigns (x : Int) : Bool := decide (x % 6 = 1 ∨ x % 6 = 2)

example : sixSigns (1 + 3) ≠ sixSigns 1 ∧
    ssCount (past sixSigns 6 6) = 2 ∧
    (6 : Int) - 6 ≤ 0 ∧ (0 : Int) < 3 ∧ (3 : Int) + 2 < 6 ∧
    sixSigns 0 = false ∧ sixSigns 3 = false ∧ sixSigns 4 = false ∧ sixSigns 5 = false ∧
    phase 3 0 = phase 3 3 := by decide

-- With p=0, periodicity is automatic and phase truncates negative clocks.
example : ∀ x : Int, sixSigns (x + 0) = sixSigns x := by
  intro x
  simp

example : ssCount (past sixSigns 0 6) = 2 ∧
    (0 : Int) - 6 ≤ -6 ∧ (-6 : Int) < -3 ∧ (-3 : Int) + 2 < 0 ∧
    sixSigns (-6) = false ∧ sixSigns (-3) = false ∧
    sixSigns (-2) = false ∧ sixSigns (-1) = false ∧
    phase 0 (-6) = phase 0 (-3) := by decide

end IndependentSSSAudit
