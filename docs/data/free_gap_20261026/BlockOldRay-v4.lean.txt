private def atomWords : List (List Bool) :=
  [[true],[false,true,true,true],[false,false,true,true,true,true,false,true,true,true]]

private def atomFacts (v : List Bool) : Prop :=
  -8≤runKey v ∧ 0≤runKey v+4*(cntF v : Int) ∧ -2≤mass v ∧
  (mass v<0 → runKey v≤0) ∧ (mass v=-2 → runKey v=-8 ∧ cntF v=2)

private instance (v : List Bool) : Decidable (atomFacts v) := by
  unfold atomFacts
  infer_instance

private theorem atom_prefix_facts (B : List Bool) (hB : B∈atomWords) (k : Nat) :
    atomFacts (B.take k) := by
  have hchecked : atomWords.all (fun B => (List.range (B.length+1)).all
      (fun k => decide (atomFacts (B.take k))))=true := by decide
  have get (q : Nat) (hq : q≤B.length) : atomFacts (B.take q) := by
    have h := List.all_eq_true.mp (List.all_eq_true.mp hchecked B hB) q (List.mem_range.mpr (by omega))
    exact of_decide_eq_true h
  by_cases hk : k≤B.length
  · exact get k hk
  · have h := get B.length (Nat.le_refl _)
    rw [List.take_of_length_le (Nat.le_refl _)] at h
    rw [List.take_of_length_le (by omega)]
    exact h

theorem runKey_append (u v : List Bool) :
    runKey (u++v)=runKey u+runKey v+4*mass u*(cntF v : Int) := by
  have hc := cntF_int v
  have hm := mass_eq v
  simp only [runKey,areaW_append,mass_append]
  grind

private theorem prefix_append_bounds (w B : List Bool) (hw : BlockWord w)
    (hB : B∈atomWords)
    (ih : ∀ k, -8≤runKey (w.take k) ∧ (mass (w.take k)<0 → runKey (w.take k)≤0)) :
    ∀ k, -8≤runKey ((w++B).take k) ∧
      (mass ((w++B).take k)<0 → runKey ((w++B).take k)≤0) := by
  intro k
  by_cases hk : k≤w.length
  · rw [List.take_append_of_le_length hk]
    exact ih k
  · rw [List.take_append,List.take_of_length_le (by omega)]
    let v := B.take (k-w.length)
    change -8≤runKey (w++v) ∧ (mass (w++v)<0 → runKey (w++v)≤0)
    obtain ⟨hlo,hplus,hmv,hneg,hmin⟩ := atom_prefix_facts B hB (k-w.length)
    change -8≤runKey v at hlo
    change 0≤runKey v+4*(cntF v : Int) at hplus
    change -2≤mass v at hmv
    change mass v<0 → runKey v≤0 at hneg
    change mass v=-2 → runKey v=-8 ∧ cntF v=2 at hmin
    by_cases hw0 : w=[]
    · subst w
      simpa only [List.nil_append] using And.intro hlo hneg
    · have hm := (sufPos_block hw [] w rfl hw0).1
      have hlow := block_area_quadratic_lower hw
      have hP : -2≤runKey w := by unfold runKey; omega
      have hp := Int.mul_nonneg (show 0≤mass w-1 by omega) (show 0≤(cntF v : Int) by omega)
      have hid : (mass w-1)*(cntF v : Int)=mass w*(cntF v : Int)-(cntF v : Int) := by grind
      rw [hid] at hp
      have hbig : 4*(cntF v : Int)≤4*mass w*(cntF v : Int) := by
        have he : 4*mass w*(cntF v : Int)=4*(mass w*(cntF v : Int)) := by grind
        rw [he]
        omega
      constructor
      · rw [runKey_append]
        omega
      · intro hnegative
        rw [mass_append] at hnegative
        have hm1 : mass w=1 := by omega
        have hmv2 : mass v=-2 := by omega
        obtain ⟨hPv,hSv⟩ := hmin hmv2
        have hPw : runKey w=0 := by rw [block_mass_one hw hm1]; decide
        rw [runKey_append,hPw,hm1,hPv,hSv]
        decide

/-- Uniform prefix-area controls for the actual BlockWord language. -/
theorem block_prefix_key_bounds {w : List Bool} (hw : BlockWord w) :
    ∀ k, -8≤runKey (w.take k) ∧ (mass (w.take k)<0 → runKey (w.take k)≤0) := by
  induction hw with
  | nil => intro k; simp [runKey,areaW,mass]
  | free hw ih => exact prefix_append_bounds _ _ hw (by simp [atomWords]) ih
  | aas hw ih => exact prefix_append_bounds _ _ hw (by simp [atomWords]) ih
  | triple hw ih => exact prefix_append_bounds _ _ hw (by simp [atomWords]) ih

/-- A fixed separator followed by A signs preserves this suffix key. -/
def oldKey (v : List Bool) (h : Nat) : Int := runKey v+4*(h : Int)*mass v-2*(h : Int)^2

private theorem separated_key_mass (u v : List Bool) (h : Nat) (hh : 3≤h)
    (hlo : -8≤runKey u) (hneg : mass u<0 → runKey u≤0)
    (h1 : oldKey (u++v) h=-2) (h2 : oldKey v h=-2) : mass (u++v)=mass v := by
  have he : runKey u+4*(mass u*((cntF v : Int)+(h : Int)))=0 := by
    unfold oldKey at h1 h2
    rw [runKey_append,mass_append] at h1
    grind
  by_cases hm0 : mass u=0
  · rw [mass_append,hm0]
    omega
  · by_cases hmpos : 0<mass u
    · have hp := Int.mul_nonneg (show 0≤mass u-1 by omega)
        (show 0≤(cntF v : Int)+(h : Int) by omega)
      have hid : (mass u-1)*((cntF v : Int)+(h : Int))=
          mass u*((cntF v : Int)+(h : Int))-((cntF v : Int)+(h : Int)) := by grind
      rw [hid] at hp
      omega
    · have hn : mass u<0 := by omega
      have hp := Int.mul_neg_of_neg_of_pos hn
        (show 0<(cntF v : Int)+(h : Int) by omega)
      have hu := hneg hn
      omega

private theorem small_separator_false {v : List Bool} (hv : BlockWord v) (hne : v≠[])
    (h i : Nat) (hh : h≤2) (hmass : mass v+(i : Int)=(h : Int)+1)
    (hkey : oldKey v h=-2) : False := by
  have hm := (sufPos_block hv [] v rfl hne).1
  have hb := block_area_quadratic_lower hv
  have hP : -2≤runKey v := by unfold runKey; omega
  unfold oldKey at hkey
  by_cases h0 : h=0
  · subst h
    have hmass1 : mass v=1 := by omega
    rw [block_mass_one hv hmass1] at hkey
    simp [runKey,areaW,mass,Recaman.ShortPeriodicSupply.sign] at hkey
  · by_cases h1 : h=1
    · subst h
      simp at hkey
      omega
    · have h2 : h=2 := by omega
      subst h
      have hmass1 : mass v=1 := by omega
      rw [block_mass_one hv hmass1] at hkey
      simp [runKey,areaW,mass,Recaman.ShortPeriodicSupply.sign] at hkey

def oldRay (w : List Bool) (h i : Nat) : Prop :=
  ∃ t, t<w.length ∧ mass (w.drop t)+(i : Int)=(h : Int)+1 ∧ oldKey (w.drop t) h=-2

private theorem drop_nonempty (w : List Bool) (t : Nat) (ht : t<w.length) : w.drop t≠[] := by
  intro he
  have hl : (w.drop t).length=w.length-t := List.length_drop
  rw [he] at hl
  simp only [List.length_nil] at hl
  omega

private theorem ordered_old_mass {w : List Bool} (hw : BlockWord w) (h t s : Nat)
    (hh : 3≤h) (hts : t≤s) (h1 : oldKey (w.drop t) h=-2) (h2 : oldKey (w.drop s) h=-2) :
    mass (w.drop t)=mass (w.drop s) := by
  have hblock := block_suffix hw (w.take t) (w.drop t) (List.take_append_drop t w)
  have hb := block_prefix_key_bounds hblock (s-t)
  have he := List.take_append_drop (s-t) (w.drop t)
  rw [List.drop_drop,show t+(s-t)=s by omega] at he
  have hc : oldKey ((w.drop t).take (s-t)++w.drop s) h=-2 := by rw [he]; exact h1
  have hm := separated_key_mass _ _ h hh hb.1 hb.2 hc h2
  rw [he] at hm
  exact hm

/-- Only the clock is unique. Different old starts with the same mass and
area are deliberately retained. -/
theorem oldRay_clock_unique {w : List Bool} (hw : BlockWord w) (h i j : Nat)
    (hi : oldRay w h i) (hj : oldRay w h j) : i=j := by
  obtain ⟨t,ht,hmt,hkt⟩ := hi
  obtain ⟨s,hs,hms,hks⟩ := hj
  by_cases hh : 3≤h
  · by_cases hts : t≤s
    · have hm := ordered_old_mass hw h t s hh hts hkt hks
      omega
    · have hm := ordered_old_mass hw h s t hh (by omega) hks hkt
      omega
  · have hv := block_suffix hw (w.take t) (w.drop t) (List.take_append_drop t w)
    exact False.elim (small_separator_false hv (drop_nonempty w t ht) h i (by omega) hmt hkt)
